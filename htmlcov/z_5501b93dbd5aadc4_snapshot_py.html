18def later_on_item(history, record): 

19 """A later lease or reset on the same item means the loop moved on.""" 

20 return any(other["id"] > record["id"] and other["assignment"] == record["assignment"] 

21 and other["kind"] in {"lease", "reset"} for other in history)

24def ask(outcome, lease=None): 

25 """What a stopped run asks of a person: its outcome's actions and options.""" 

26 source = outcome or lease 

27 if source is None: 

28 return None 

29 return {"agent": source["agent"], 

30 "outcome": (outcome or {}).get("outcome") or (lease or {}).get("result") or "", 

31 "summary": (outcome or {}).get("summary") or (lease or {}).get("summary") or "", 

32 "actions": (outcome or {}).get("actions", []), "options": (outcome or {}).get("options", []), 

33 "created": source["created"], "url": source["url"]}

36class MonitorSource: 

37 def __init__(self, config, github, coordinator, clock): 

38 self.config = config 

39 self.github = github 

40 self.trust = coordinator.trust 

41 self.coordinator = coordinator 

42 self.clock = clock 

43 self.titles = {} 

44 self.planning = EligiblePlanning(config, github, coordinator, clock)

46 def snapshot(self): 

47 now = self.clock() 

48 comments = self.github.repository_comments(lookback_seconds=WINDOW_SECONDS) 

49 history, invalid = self.coordinator.repository_history(comments) 

50 renewed = {comment["id"]: comment.get("updated_at") for comment in comments} 

51 notices = [f"Unreadable coordination records on #{number}" for number in sorted(invalid)] 

52 newest = max((record_version(comment.get("body")) or 0 for comment in comments), default=0) 

53 if newest > record_version(MARKER): 

54 notices.append(f"Some records use format v{newest}, newer than this ub-agents reads: upgrade it.") 

55 open_items = {item.number: item for item in self.github.observe(details=False)} 

56 self.titles.update((number, item.title) for number, item in open_items.items()) 

57 leases = [r for r in history if r["kind"] == "lease"] 

58 labelled = self.labelled(open_items, notices) 

59 stopped = {entry["item"]["number"] for entry in labelled} 

60 blocked = [lease for lease in leases if lease.get("result") == "blocked" 

61 and lease["assignment"] in open_items and lease["assignment"] not in stopped 

62 and not later_on_item(history, lease)] 

63 eligible = self.planning.snapshot() 

64 return { 

65 "repository": self.config.repository, 

66 "folder": str(self.config.root), 

67 "local_host": socket.gethostname(), 

68 "stop_labels": list(self.config.stop_labels), 

69 "stop_outcomes": self.stop_outcomes(), 

70 "generated_at": iso(now), 

71 "notices": notices, 

72 "needs_human": labelled + [{"item": self.item(lease), "labels": ["blocked"], "labelled": False, 

73 "ask": ask(self.outcome(history, lease), lease), "notice": None} 

74 for lease in blocked], 

75 "running": [self.run(lease, renewed) for lease in live_leases(history, now)], 

76 "eligible": eligible, 

77 "eligible_error": self.planning.error, 

78 "expired": [self.run(lease, renewed) for lease in leases if lease["state"] in CLAIMED 

79 and seconds(lease["expires"]) <= now and not later_on_item(history, lease)], 

80 "recent": sorted((self.ended(history, lease, renewed) for lease in leases 

81 if lease["state"] == "released" and seconds(lease["expires"]) >= now - WINDOW_SECONDS), 

82 key=lambda run: run["ended"], reverse=True), 

83 "rates": self.rates(), 

84 }

86 def stop_outcomes(self): 

87 """The stop outcomes and the agents that declare each one.""" 

88 stops = set(self.config.stop_labels) 

89 outcomes = {} 

90 for agent in self.config.agents: 

91 for name, changes in agent.outcomes.items(): 

92 if stops.intersection(changes.get("add", ())): 

93 outcomes.setdefault(name, []).append(agent.name) 

94 return [{"name": name, "agents": sorted(agents)} for name, agents in sorted(outcomes.items())]

96 def labelled(self, open_items, notices): 

97 """Open items with a stop label, each with the ask of its latest outcome.""" 

98 entries, trusted = [], self.trust.observation() 

99 for number, item in sorted(open_items.items()): 

100 labels = sorted(item.labels.intersection(self.config.stop_labels)) 

101 if not labels: 

102 continue 

103 comments = self.github.comments(number) 

104 try: 

105 history = records(comments, trusted=trusted) 

106 except RecordError: 

107 notices.append(f"Unreadable coordination records on #{number}") 

108 history = [] 

109 outcomes = [r for r in history if r["kind"] == "outcome"] 

110 released = [r for r in history if r["kind"] == "lease" and r["state"] == "released"] 

111 notice = next((c for c in reversed(comments) if (c.get("body") or "").startswith(ACTION_MARKER) 

112 and trusted(c.get("user"))), None) 

113 entries.append({"item": self.reference(number, item.kind), "labels": labels, "labelled": True, 

114 "ask": ask(outcomes[-1] if outcomes else None, released[-1] if released else None), 

115 "notice": notice.get("html_url") if notice else None}) 

116 return entries

118 @staticmethod 

119 def outcome(history, lease): 

120 return next((r for r in reversed(history) if r["kind"] == "outcome" 

121 and r["lease_id"] == lease["id"] and r["agent"] == lease["agent"]), None)

123 def reference(self, number, kind): 

124 path = "pull" if kind == "pr" else "issues" 

125 return {"number": number, "kind": kind, "title": self.title(number, kind), 

126 "url": f"https://github.com/{self.config.repository}/{path}/{number}"}

128 def title(self, number, kind): 

129 # A closed item's title is read once per monitor; it rarely changes. 

130 if number not in self.titles: 

131 self.titles[number] = self.github.item(number, kind if kind == "pr" else None).title 

132 return self.titles[number]

134 def item(self, lease): 

135 # The launcher pins a PR's head in its lease; an issue has none. 

136 return self.reference(lease["assignment"], "pr" if lease.get("assignment_sha") else "issue")

138 def run(self, lease, renewed): 

139 return {"id": lease["id"], "agent": lease["agent"], "item": self.item(lease), "host": lease.get("host") or "", 

140 "actor": lease["actor"], "runtime": lease["runtime"], "attempt": lease["attempt"], 

141 "state": lease["state"], "started": lease["created"], "renewed": renewed.get(lease["id"]), 

142 "expires": lease["expires"], "url": lease["url"]} 

143 

144 def ended(self, history, lease, renewed): 

145 outcome = self.outcome(history, lease) 

146 usage = lease.get("usage") or {} 

147 return self.run(lease, renewed) | { 

148 "ended": lease["expires"], "status": lease.get("result") or "released", 

149 "result": (outcome or {}).get("outcome") or lease.get("result") or "released", 

150 "summary": lease_summary(history, lease), "tokens": usage.get("tokens"), 

151 "cost": usage.get("total_cost_usd")}

153 def rates(self): 

154 # GitHub counts REST and GraphQL calls in separate hourly buckets, and 

155 # every tool using the same account draws from them. 

156 result = {} 

157 for resource, headers in getattr(self.github, "resource_quotas", {}).items(): 

158 try: 

159 result[resource] = {key: int(headers[f"x-ratelimit-{key}"]) for key in ("remaining", "limit", "reset")} 

160 except (KeyError, TypeError, ValueError): 

161 continue 

162 return result | {"counted": getattr(self.github, "quota_requests", 0), 

163 "graphql_calls": getattr(self.github, "graphql_calls", 0), 

164 "not_modified": getattr(self.github, "not_modified_responses", 0)}
