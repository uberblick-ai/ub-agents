10def arguments(commands): 

11 retry = commands.add_parser("retry", help="let stopped work run again, with a recorded reason", 

12 description="Record a human-authorized reset of blocked work and attempt limits. " 

13 "Use after resolving the cause; on open items it then removes stop labels and restores the agent's trigger. " 

14 "Without --agent, print and use the first configured agent whose kind applies and whose trigger is on the item. " 

15 "If none has a trigger on the item, use the first agent whose kind applies.",

18 retry.add_argument("--agent", metavar="NAME", 

19 help="configured agent (default: first matching item kind and trigger, else first matching kind)") 

20 retry.add_argument("--reason", required=True, help="record why the human-authorized reset is justified")

23 help="issue or PR number")

28def retry_labels(github, item, agent, stop_labels): 

29 if item.state == "closed": 

30 return f"#{item.number} is closed." 

31 stops = [label for label in stop_labels if label in item.labels] 

32 added = [] if any(label in item.labels for label in agent.triggers) else [agent.triggers[0]] 

33 remaining = list(stops) 

34 try: 

35 for label in stops: 

36 github.remove_label(item.number, label) 

37 remaining.remove(label) 

38 if added: 

39 github.add_labels(item.number, added) 

40 except AgentError as exc: 

41 steps = [] 

42 if remaining: 

43 steps.append(f"remove {', '.join(remaining)}") 

44 if added: 

45 steps.append(f"add {', '.join(added)}") 

46 raise AgentError(f"Reset recorded, but label changes failed ({exc}). " 

47 f"By hand, {'; '.join(steps)}.") from exc 

48 changes = [] 

49 if stops: 

50 changes.append(f"removed {', '.join(stops)}") 

51 if added: 

52 changes.append(f"added {', '.join(added)}") 

53 message = ', '.join(changes) if changes else "Labels were already in place" 

54 return f"{message[0].upper()}{message[1:]}. A running launcher picks it up on its next poll."

57@command("retry", arguments, number="required", legacy_number=True, config_required=True, 

58 build=("config", "github", "coordinator")) 

59def retry_run(args, config, github, actor, coordinator): 

60 if args.agent is None: 

61 item = github.item(args.number) 

62 matching = [agent for agent in config.agents if agent.kind in {item.kind, "either"}] 

63 if not matching: 

64 raise AgentError(f"No configured agent applies to {item.kind} #{item.number}") 

65 agent = next((agent for agent in matching if item.labels.intersection(agent.triggers)), matching[0]) 

66 args.agent = agent.name 

67 print(f"Using agent {agent.name} for {item.kind} #{item.number}.") 

68 agent = next((agent for agent in config.agents if agent.name == args.agent), None) 

69 if agent is None: 

70 raise AgentError("Unknown configured agent") 

71 if not args.reason.strip(): 

72 raise AgentError("retry requires a positive item number and a reason") 

73 history = coordinator.history(args.number) 

74 if live_leases(history, timestamp()): 

75 raise AgentError("Cannot reset attempts while an assignment is owned") 

76 item = github.item(args.number) 

77 record = {"kind": "reset", "agent": args.agent, "actor": actor, 

78 "assignment": item.number, "created": iso(timestamp()), "summary": args.reason} 

79 created = records([github.create_comment(item.number, body(record))])[0] 

80 coordinator.notices.resumed(item.number) 

81 print(f"Reset {args.agent} attempts on #{item.number}: {created['url']}") 

82 print(retry_labels(github, item, agent, config.stop_labels)) 

83 return
