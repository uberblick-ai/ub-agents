1"""Capture runtime streams separately while retaining the combined process log."""

3from contextlib import contextmanager, ExitStack 

4import os 

5import subprocess

8class ProcessOutput: 

9 def __init__(self, log, stdout, stderr, readers=()): 

10 self.log, self.stdout, self.stderr, self.readers = log, stdout, stderr, readers 

11 

12 def poll(self): 

13 for reader in self.readers: 

14 # Bound this copy to bytes already written so a noisy child cannot 

15 # keep supervision from checking shutdown and ownership deadlines. 

16 remaining = max(0, os.fstat(reader.fileno()).st_size - reader.tell()) 

17 while remaining: 

18 chunk = reader.read(min(remaining, 65536)) 

19 if not chunk: 

20 break 

21 self.log.write(chunk) 

22 remaining -= len(chunk) 

23 self.log.flush()

26@contextmanager 

27def capture_streams(run_dir, separate=False): 

28 with ExitStack() as stack: 

29 log = stack.enter_context((run_dir / "process.log").open("wb")) 

30 if separate: 

31 paths = [run_dir / name for name in ("stdout.log", "stderr.log")] 

32 writers = [stack.enter_context(path.open("wb", buffering=0)) for path in paths] 

33 readers = [stack.enter_context(path.open("rb")) for path in paths] 

34 output = ProcessOutput(log, *writers, readers) 

35 else: 

36 output = ProcessOutput(log, log, subprocess.STDOUT) 

37 try: 

38 yield output 

39 finally: 

40 output.poll()
