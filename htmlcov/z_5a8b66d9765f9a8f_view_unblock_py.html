13ACTION_MARKER = '<!-- ub-agents:action-needed ' 

14LINKS = re.compile(r'^\[Claim\]\(.*?\) · (?:\[Outcome\]\(.*?\)|No outcome was reported\.)$')

17def comment_body(value): 

18 lines = description_text(value).splitlines() 

19 if lines and lines[0].startswith(ACTION_MARKER): 

20 lines.pop(0) 

21 if lines and lines[0] == '**Action needed**': 

22 lines.pop(0) 

23 link = next((index for index, line in enumerate(lines) if LINKS.fullmatch(line)), None) 

24 if link is not None and '<summary>Reasoning and evidence</summary>' not in lines: 

25 lines.pop(link) 

26 return '\n'.join(lines).strip()

29def comment_sections(body): 

30 """Fold the generated notice wrapper, leaving nested supporting Markdown intact.""" 

31 wrapper = re.search(r'<details>\n<summary>(?:Reasoning and evidence|Reasoning, evidence and resume instructions)</summary>\n\n', body) 

32 if not wrapper: 

33 return body, '' 

34 visible, details = body[:wrapper.start()], body[wrapper.end():] 

35 if details.endswith('\n\n</details>'): 

36 details = details[:-len('\n\n</details>')] 

37 return visible.strip(), details.strip()

40def resume_section(body): 

41 """Extract same-role workflow steps, leaving approval-gate steps visible.""" 

42 wrapper = re.search(r'<details>\n<summary>(To send it back to [^<\n]+ instead|To resume [^<\n]+)</summary>\n\n', body) 

43 if not wrapper: 

44 tail = re.search(r'(?m)^Then resume ([^:\n]+):\n\n', body) 

45 if tail: 

46 return body[:tail.start()].strip(), f'To resume {tail[1]}', body[tail.end():].strip() 

47 return body, '', '' 

48 steps = body[wrapper.end():].rstrip() 

49 if steps.endswith('\n\n</details>'): 

50 steps = steps[:-len('\n\n</details>')] 

51 return body[:wrapper.start()].strip(), wrapper[1], steps.strip()

54def trust_reason(author, authors): 

55 if not isinstance(author, str) or not author: 

56 return 'The comment author is unreadable and cannot be verified.' 

57 verified = mapping(authors.get(author.casefold())) 

58 if verified.get('trusted') is True: 

59 return '' 

60 return text(verified.get('reason'), '') or f'The launcher has not verified @{author} for coordination records.'

71 def details(self, now=None, pending=False): 

72 now = time.time() if now is None else now 

73 lines = [] if self.available else ['No action-needed comment is cached; press g to load from GitHub.'] 

74 if not self.available and self.omitted: 

75 suffix = 'loading from GitHub…' if pending else 'press g to load from GitHub.' 

76 lines = ['Comment left out of the snapshot to save space; ' + suffix] 

77 if self.error: 

78 lines.append(self.error + ' Press g to retry.') 

79 if self.notice: 

80 lines.append(self.notice) 

81 if self.source: 

82 created = stamp(self.created_at) 

83 clock = datetime.fromtimestamp(created).astimezone().strftime('%H:%M') if created is not None else None 

84 parts = ['Source: action-needed comment', clock, self.source] 

85 if self.observed_at is not None: 

86 parts.append(f'loaded {max(0, now - self.observed_at):.0f}s ago') 

87 lines.append(' · '.join(part for part in parts if part)) 

88 return '\n'.join(lines)

91def needs_attention(row): 

92 if not row or row.hidden or row.group != 'Needs attention': 

93 return False 

94 if row.attention_rows: 

95 return any(needs_attention(agent) for agent in row.attention_rows) 

96 failures, maximum = row.data.get('failures'), row.data.get('max_attempts') 

97 exhausted = type(failures) is int and type(maximum) is int and failures >= maximum 

98 return row.state in {'parked', 'blocked'} or row.state == 'failed' and exhausted

102 notice = mapping(mapping(session.data.get('action_needed')).get(str(row.item))) if row and session else {} 

103 if notice.get('omitted') is True:

116def plan_reason(row): 

117 """Remove observer-added notice text and generated same-role retry prose.""" 

118 reason = description_text(row.reason) 

119 summary = description_text(text(row.data.get('attention_reason'), '')) 

120 prefix = re.match(r'^(?:Last run blocked|Stop label .+? is present): ', reason) 

121 start = prefix.end() if prefix else 0 

122 if summary and reason[start:].endswith(': ' + summary): 

123 reason = reason[:-len(': ' + summary)] 

124 return re.sub(r'; inspect (?:outcome|failures) and use ub-agents retry \d+ --agent \S+ --reason TEXT$', '', reason)

127def other_role_context(row, body): 

128 """Show other causes without the notice summary appended by the observer.""" 

129 visible, supporting = comment_sections(body) 

130 lead, title, _ = resume_section(visible) 

131 role = (title.removeprefix('To resume ') if title.startswith('To resume ') else 

132 title.removeprefix('To send it back to ').removesuffix(' instead')) 

133 reasons = {notice_summary(part) for part in (lead, supporting) if part} 

134 sections = [] 

135 for agent in row.attention_rows: 

136 if agent.agent == role: 

137 continue 

138 reason = short_reason(plan_reason(agent).removeprefix('Last run blocked: ')) 

139 if reason and reason not in reasons: 

140 sections.append(f'### {agent.agent} · {attention_state(agent)[1]}\n\n{notice_reason(reason)}') 

141 return lead, '\n\n'.join(sections)

144def fallback_body(row): 

145 if row and len(row.attention_rows) > 1: 

146 bodies, resumes = [], [] 

147 for agent in row.attention_rows: 

148 body = fallback_body(agent) 

149 lead, _, resume = resume_section(body) 

150 if lead: 

151 bodies.append(f'### {agent.agent} · {attention_state(agent)[1]}\n\n{lead}') 

152 if resume: 

153 resumes.append(f'### {agent.agent}\n\n{resume}') 

154 body = '\n\n'.join(bodies) 

155 if resumes: 

156 title = ('To send it back to these agents instead' if row.data.get('kind') == 'pr' 

157 else 'To resume these agents') 

158 body += (f'\n\n<details>\n<summary>{title}</summary>\n\n' + 

159 '\n\n'.join(resumes) + '\n\n</details>') 

160 return body 

161 context = plan_reason(row) if row else '' 

162 summary = description_text(text(row.data.get('attention_reason'), '')) if row else '' 

163 if summary and summary not in context: 

164 context = '\n\n'.join(part for part in (context, summary) if part) 

165 if row and row.state == 'blocked' and needs_attention(row): 

166 steps = (f"```sh\n{retry_command(row.item, row.agent)}\n```\n\n" 

167 "Use these steps only when resuming the same role; follow the project's " 

168 "correction or handoff route if a different role must act next.") 

169 resume = outcome_resume(row.item, text(row.agent), steps, is_pr=row.data.get('kind') == 'pr') 

170 return f"{context}\n\n{resume}".strip() 

171 return context.strip()

174def unblock_body(row, comment): 

175 if comment.available: 

176 if row and len(row.attention_rows) > 1: 

177 lead, context = other_role_context(row, comment.body) 

178 if context: 

179 return lead + '\n\n' + context + comment.body[len(lead):] 

180 return comment.body 

181 body = fallback_body(row) 

182 return '**Fallback: action-needed comment unavailable.**\n\n' + body if body else ''

185def unblock_metadata(row, comment, session, now=None): 

186 if not row: 

187 return '', '' 

188 _, state = attention_state(row) 

189 parts = [attention_agents(row)] if len(row.attention_rows) > 1 else [row.agent, state] 

190 start = row.waiting_since() 

191 created = stamp(start) 

192 waiting = '' 

193 if created is not None: 

194 waiting = 'waiting ' + waiting_time(start, now) 

195 parts.extend([waiting, 'since ' + datetime.fromtimestamp(created).astimezone().strftime('%H:%M')])
