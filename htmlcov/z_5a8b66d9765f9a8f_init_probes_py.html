20@dataclass 

21class Group: 

22 name: str 

23 summary: list[str] = field(default_factory=list) 

24 problems: list = field(default_factory=list)

26 @property 

27 def status(self): 

28 statuses = {check.status for check in self.problems} 

29 return "fail" if statuses & {"fail", "skip"} else "warn" if statuses else "ok"

32class Probes: 

33 def __init__(self, plan, github, doctor=None): 

34 self.plan, self.github = plan, github 

35 self.doctor = doctor or Doctor(github=github) 

36 self.groups = [] 

37 self.clis = {} 

38 self.existing_labels = None 

39 self.discussions = None 

40 self.discussions_off = False 

41 self.discussions_note = "" 

42 self.branch = self.default_branch = self.clean = None 

43 self.ub = False 

44 self.actor = None 

45 self.teams = [] 

46 self.approvals = "" 

47 self.visibility = None

49 def collect(self, name, summary=()): 

50 group = Group(name, [part for part in summary if part]) 

51 group.problems = [check for check in self.doctor.checks if check.status != "ok"] 

52 self.doctor.checks.clear() 

53 self.groups.append(group) 

54 return group

56 def run(self): 

57 self.doctor.checks.clear() 

58 self.groups = [] 

59 git_ready, gh_ready = self._machine() 

60 actor, metadata = self._github(gh_ready) 

61 self._checkout(git_ready, metadata) 

62 self._agents() 

63 self._account(actor) 

64 

65 if metadata: 

66 try: 

67 self.existing_labels = {name.casefold() for name in self.github.labels()} 

68 except PROBE_ERRORS: 

69 pass 

70 self.check_discussions() 

71 else: 

72 self.discussions_note = "GitHub is unavailable" 

73 return self

75 def _machine(self): 

76 d, config = self.doctor, self.plan.config 

77 python = f"Python {sys.version_info.major}.{sys.version_info.minor}" 

78 d.add("platform", "ok" if sys.platform in {"darwin", "linux"} else "fail", sys.platform, "Use macOS or Linux") 

79 git_ready = d.tool("git", "Install Git and put git on PATH") 

80 gh_ready = d.tool("gh", "Install GitHub CLI from https://cli.github.com") 

81 try: 

82 rows = parse_process_table(d.probe(["ps", "-axo", "pid=,pgid=,stat="])) 

83 if os.getpid() not in {pid for pid, _, _ in rows}: 

84 raise AgentError("missing process") 

85 except PROBE_ERRORS: 

86 d.add("process-inspection", "fail", "ps can't inspect processes", "Install or allow ps") 

87 parent = config.root 

88 while not parent.exists(): 

89 parent = parent.parent 

90 if not os.access(parent, os.W_OK | os.X_OK): 

91 d.add("local-state", "fail", "The setup directory is not writable", "Run init in a writable checkout") 

92 self.collect("Machine", [python, "git" if git_ready else "", "gh" if gh_ready else "", "ps"]) 

93 return git_ready, gh_ready

95 def _github(self, gh_ready): 

96 d, config = self.doctor, self.plan.config 

97 actor = metadata = None 

98 if gh_ready: 

99 try: 

100 actor = self.github.actor() 

101 except PROBE_ERRORS: 

102 d.add("github-auth", "fail", "gh is not logged in", "Run gh auth login") 

103 if actor: 

104 metadata = d.repository_access(config, self.github) 

105 github = self.collect("GitHub", [actor, config.repository]) 

106 if metadata and not github.problems: 

107 github.summary.append("labels writable") 

108 try: 

109 self.visibility = repository_visibility(metadata) 

110 except AgentError: 

111 self.visibility = None 

112 try: 

113 value, source = resolve_policy(config.approvals, lambda: repository_visibility(metadata)) 

114 self.approvals = f"Approvals: {value} ({source})" 

115 except AgentError: 

116 self.approvals = "Approvals: unresolved (repository visibility is unreadable)" 

117 return actor, metadata

119 def _checkout(self, git_ready, metadata): 

120 d, config = self.doctor, self.plan.config 

121 if git_ready: 

122 read_git = lambda root, *args: d.probe(["git", "-C", str(root), *args]).strip() 

123 branch, clean = None, None 

124 for id, error in repository_checks(config, read_git, d.runner): 

125 if error: 

126 d.add(id, "fail", str(error), "Run init at the root of a checkout whose origin is the repository") 

127 try: 

128 branch = read_git(config.root, "rev-parse", "--abbrev-ref", "HEAD") 

129 clean = not read_git(config.root, "--no-optional-locks", "status", "--porcelain=v1", "--untracked-files=all") 

130 except PROBE_ERRORS: 

131 d.add("branch", "fail", "This is not a Git checkout", "Run init inside a clone of the repository") 

132 default = metadata.get("default_branch") if metadata else None 

133 self.branch, self.default_branch = branch, default if isinstance(default, str) and default else None 

134 synced = False 

135 if isinstance(default, str) and default and branch: 

136 # Flow handles the branch question and stops setup on local changes. 

137 problems = [error for id, error in control_checkout_checks(config, default, read_git) 

138 if error and id not in {"control-checkout-branch", "control-checkout-dirty"}] 

139 for error in problems: 

140 d.add("control-checkout", "warn", str(error), "Fine for setup; launch from a clean default-branch checkout", 

141 required=False) 

142 synced = not problems 

143 self.clean = clean 

144 self.collect("Checkout", [branch, "clean" if clean else "local changes" if clean is False else "", 

145 "in sync with origin" if synced else ""])

147 def _agents(self): 

148 d = self.doctor 

149 for cli in CLIS: 

150 if not d.which(cli): 

151 self.clis[cli] = "not installed" 

152 continue 

153 try: 

154 d.probe(AUTH_PROBES[cli]) 

155 self.clis[cli] = "logged in" 

156 except PROBE_ERRORS: 

157 self.clis[cli] = "not logged in" 

158 if all(state == "not installed" for state in self.clis.values()): 

159 d.add("runtime", "fail", "Neither claude nor codex is installed", "Install Claude Code or Codex before launch") 

160 elif "logged in" not in self.clis.values(): 

161 d.add("runtime-login", "warn", "No agent CLI is logged in", "Run claude auth login or codex login before launch", 

162 required=False) 

163 self.collect("Agents", [f"{cli} {state}" for cli, state in self.clis.items()]) 

164 self.ub = bool(d.which("ub"))

166 def _account(self, actor): 

167 d, config = self.doctor, self.plan.config 

168 if actor: 

169 try: 

170 role = Roles(self.github)({"login": actor}) 

171 except PROBE_ERRORS: 

172 role = None 

173 if role in {"maintain", "admin"}: 

174 d.add("account", "warn", f"{actor} is {role}, good for setup; launch the loop with a write-access agent account", 

175 f"See {ACCOUNT_GUIDANCE}", required=False) 

176 elif role is None: 

177 d.add("account", "warn", f"{actor}'s repository role can't be read", f"See {ACCOUNT_GUIDANCE}", required=False) 

178 elif role != "write": 

179 d.add("account", "fail", f"{actor} can't push to {config.repository}", f"See {ACCOUNT_GUIDANCE}") 

180 if config.launchers is not None and actor.casefold() not in {v.casefold() for v in config.launchers}: 

181 d.add("launcher-listed", "warn", f"{actor} is not in local launchers; it will claim no work", 

182 "Review ub-agents.local.yaml", required=False) 

183 self.actor = actor 

184 try: 

185 self.teams = self.github.teams() 

186 except PROBE_ERRORS: 

187 self.teams = [] 

188 account = self.collect("Account") 

189 if not account.problems: 

190 self.groups.remove(account)

192 def check_discussions(self): 

193 from .init_boards import Boards 

194 try: 

195 repo = Boards(self.github, self.plan.config.repository).discussions() 

196 except PROBE_ERRORS: 

197 self.discussions_note = "Discussions can't be checked" 

198 return 

199 if repo["hasDiscussionsEnabled"] is not True: 

200 self.discussions_off = True 

201 self.discussions_note = f"Discussions are off in {self.plan.config.repository}" 

202 elif repo["viewerPermission"] not in {"WRITE", "MAINTAIN", "ADMIN"}: 

203 self.discussions_note = "this account can't create discussions" 

204 elif not repo["discussionCategories"]["nodes"]: 

205 self.discussions_note = "the repository has no discussion categories" 

206 else: 

207 self.discussions = repo

209 def runtime(self, preferred=None): 

210 """The CLI to preselect: an explicit choice, else the logged-in one (Codex when both).""" 

211 if preferred: 

212 return preferred 

213 ready = [cli for cli in ("codex", "claude") if self.clis.get(cli) == "logged in"] 

214 installed = [cli for cli in ("codex", "claude") if self.clis.get(cli, "not installed") != "not installed"] 

215 return (ready or installed or ["codex"])[0]
