13 retrospective = commands.add_parser("retrospective", help="post to the agent's retrospective board", overview_hidden=True, 

14 description="Post a body file as a top-level comment on the agent's configured " 

15 "retrospective discussion. Only available inside a supervised run; use the "

18 "feedback about ub-agents to its fixed Field reports board instead; omit " 

19 "project details and item links. Prints the comment URL and does not change " 

20 "the run's outcome. Upstream GitHub failures warn and exit successfully.", 

21 examples=("ub-agents retrospective --body-file /run/scratch/retrospective.md", 

22 'ub-agents retrospective --body-file "/run/scratch/run notes.md"')) 

23 retrospective.add_argument("--body-file", required=True, metavar="PATH", help="path to a UTF-8 retrospective body file") 

24 retrospective.add_argument("--upstream", action="store_true", help="send public feedback about ub-agents (requires upstream-retrospectives: on)")
